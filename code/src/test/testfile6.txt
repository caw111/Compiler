/*
 * A2: A级短路求值
 */

int calls = 0;

int hit() {
    calls = calls + 1;
    return 1;
}

int main() {
    printf("24373168\n");
    
    int score = 0;

    /*
     * 左边为 0：
     * hit() 不应执行
     */
    if (0 && hit())
        score = score + 100;

    /*
     * 左边为 1：
     * hit() 不应执行
     */
    if (1 || hit())
        score = score + 1;

    /*
     * && 左边为真：
     * hit() 必须执行
     */
    if (1 && hit())
        score = score + 2;

    /*
     * || 左边为假：
     * hit() 必须执行
     */
    if (0 || hit())
        score = score + 4;

    /*
     * && 先算：
     * 0 && hit() 短路，不调用 hit()；
     * 然后 0 || 1 为真。
     */
    if (0 && hit() || 1)
        score = score + 8;

    /*
     * || 左边已经为真，
     * 后面的 hit() && hit() 整体不求值。
     */
    if (1 || hit() && hit())
        score = score + 16;

    printf("%d %d\n", calls, score);

    return 0;
}
